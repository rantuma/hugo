waitServer

mcpcall search_docs '{"query":"intro"}' '"url":"https://example.org/p1/"'
! mcpcall search_docs '{"query":"intro"}' 'warning'

replace hugo.toml 'baseURL = "https://example.org/"' 'baseURL = "https://example.org/'
mcpcall search_docs '{"query":"intro"}' '"url":"https://example.org/p1/"' '"warning":"The last build failed, so the content may be outdated: '

replace content/p1.md 'The intro.' 'The changed intro.'

mcpcall search_docs '{"query":"changed"}' '"url":"https://example.org/p1/"'
